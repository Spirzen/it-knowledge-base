## Проверка «на глаз»: больно ли читать код?

Перед опросом — необязательное, но показательное упражнение. Больно ли вам читать такой фрагмент?

```javascript
const createProcessor = (config = { timeout: 1000, retries: 3 }) => {
  const cache = new WeakMap();
  const pending = new Map();

  return (endpoint) => async (...args) => {
    const { threshold = 0.5, transform = (v) => v } = args[0] || {};
    const key = JSON.stringify({ endpoint, args: [...args] });

    if (pending.has(key)) {
      return pending.get(key);
    }

    const executor = async (attempt = 1) => {
      try {
        const result = await new Promise((res, rej) => {
          const timer = setTimeout(() => rej(new Error('Timeout')), config.timeout);

          fetch(endpoint, { method: 'POST', body: JSON.stringify(args) })
            .then(r => r.json())
            .then(data => {
              clearTimeout(timer);
              if (data.confidence < threshold) {
                rej(new Error(`Low confidence: ${data.confidence}`));
              } else {
                res(transform(data.payload));
              }
            })
            .catch(rej);
        });

        cache.set({ endpoint, args }, result);
        return result;

      } catch (err) {
        if (attempt < config.retries) {
          const delay = 2 ** attempt * 100;
          await new Promise(r => setTimeout(r, delay));
          return executor(attempt + 1);
        }
        throw err;
      } finally {
        pending.delete(key);
      }
    };

    const promise = executor();
    pending.set(key, promise);
    return promise;
  };
};
```

## Практика руками (по желанию)

Опрос проверяет понимание. Дополнительно можно выполнить на своём ПК:

**Терминал.** Создайте `hello.txt` с текстом `Hello, world!` через терминал и проверьте содержимое (`cat` / `type`).

**Git.** `git init` → файл `readme.md` с `# Старт` → `git add` → `git commit -m "Старт"`. Разбор — [лабораторная шпаргалка Git](/lab/Примеры/1123#git-add-git-commit-git-push--пример-для-начинающих).

**Код.** Функция `greet(name)` → `Привет, {name}!`; цикл 1..10; `if age >= 18` для порога голосования.

Подробные эталоны — в обратной связи опроса и в главах по [Git](/encyclopedia/4-code-dev/4-13-osnovy-raboty-s-git/intro) и [основам кода](/encyclopedia/4-code-dev/4-02-chto-takoe-kod-i-kak-on-rabotaet/intro).
